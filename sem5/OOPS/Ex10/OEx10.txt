      count=-1;
      flag=0;
   }
   void SetNum(){
      cout<<"Enter No. of elements: "<<endl;
      cin>>n;
   }
   void Push(int x){
      try{
	 if(count<n-1){
	    Arr[++count]=x;
	    flag=1;
	 }
	 else
	    throw x;
      }
      catch(int y){
	 cout<<"Error:StackOverflow"<<endl;	 
      }    
   }
   void Pop(){
      try{
	 if (count>=0){
	    cout<<"Popped element:"<<Arr[count]<<endl;
	    Arr[count]='\0';
	    count-=1;
	 }
	 else
	    throw count;
      }
      catch (int y){
	    if(y<0 && flag==1){
	       cout<<"Error:StackUnderflow"<<endl;
	    }
	    if(y<0 && flag==0){
	       cout<<"Error:EmptyStack"<<endl;
	       throw;
	    }
	 }
   }
};
